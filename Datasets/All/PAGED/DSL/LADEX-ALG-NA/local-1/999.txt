init(i1)
action(n1, "start course packet process")
action(n2, "run: Send Course Packet Information to Faculty")
action(n3, "run: Send Design to Faculty")
action(n4, "run: Faculty chooses the design")
action(n5, "run: Identify content requiring copyright clearance and contact publisher")
action(n6, "condition: Entire Course Packet Copyright Cleared?")
branch(b1, XOR)
action(n7, "run: Contact Faculty to inform of denial and ask if new content should be added")
action(n8, "run: Calculate the Course Packet Cost")
action(n9, "run: Add Bookstore margin to determine the retail price")
action(n10, "condition: Does Faculty need to reconsider course packet content?")
branch(b2, XOR)
action(n11, "run: Notify Bookstore of Course packet  Course number  Faculty name  and sections")
action(n12, "run: Notify Packet information to the Bookstore")
action(n13, "end: Process completed successfully")
end(e1)
defines(v1, "condition: Entire Course Packet Copyright Cleared?", {"Contact Faculty to inform of denial and ask if new content should be added", "Entire Course Packet Copyright Cleared"})
defines(v2, "condition: Does Faculty need to reconsider course packet content?", {"Price of the Course Packet is finalized"})
actor(a_default, "Default", [i1, n1, n2, n3, n4, n5, n6, b1, n7, n8, n9, n10, b2, n11, n12, n13, e1])

precedence(i1, n1)
precedence(n6, b1)
precedence(n10, b2)
precedence(n13, e1)
precedence(n1, n2)
precedence(n2, n3)
precedence(n3, n4)
precedence(n4, n5)
precedence(b2, n5)
precedence(n5, n6)
precedence(n7, n6)
precedence(b1, n7)
precedence(b1, n8)
precedence(n8, n9)
precedence(n9, n10)
precedence(b2, n11)
precedence(n11, n12)
precedence(n12, n13)
guard(b1, n7, v1 = 'Contact Faculty to inform of denial and ask if new content should be added')
guard(b1, n8, v1 = 'Entire Course Packet Copyright Cleared')
guard(b2, n11, v2 = 'Price of the Course Packet is finalized')
